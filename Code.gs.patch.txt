// เพิ่มใน Code.gs ของระบบเดิม — ไม่ต้องลบฟังก์ชันเดิม
//
// เปลี่ยน doGet() เดิมเป็น:
function doGet(e) {
  if (e && e.parameter && e.parameter.bridge === "1") {
    return HtmlService.createHtmlOutputFromFile("ApiBridge")
      .setTitle("Apps Script API Bridge")
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag("viewport", "width=device-width, initial-scale=1");
  }
  return HtmlService.createHtmlOutputFromFile("Index")
    .setTitle("ระบบโรงกลึงช่างมนูญบ่อทอง")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag("viewport", "width=device-width, initial-scale=1");
}
